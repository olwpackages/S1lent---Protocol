cmake_minimum_required(VERSION 3.24)
project(S1lentUniversalIP VERSION 0.1.0 LANGUAGES CXX)

option(S1LENT_BUILD_TESTS "Build S1lent tests and local integration harness" ON)
option(S1LENT_BUILD_FUZZERS "Build libFuzzer parser targets (Clang only)" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto SSL)
if(WIN32)
  add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)
endif()

add_library(s1lent
  src/packet.cpp src/crypto.cpp src/session.cpp src/route.cpp src/udp_transport.cpp src/node.cpp src/handshake.cpp src/windows_tun.cpp src/tunnel.cpp)
target_include_directories(s1lent PUBLIC include)
target_link_libraries(s1lent PUBLIC OpenSSL::Crypto OpenSSL::SSL)
if(WIN32)
  target_link_libraries(s1lent PUBLIC ws2_32)
endif()
if(MSVC)
  target_compile_options(s1lent PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(s1lent PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()

add_executable(s1lent-client apps/client.cpp)
target_link_libraries(s1lent-client PRIVATE s1lent)
add_executable(s1lent-node apps/node_main.cpp)
target_link_libraries(s1lent-node PRIVATE s1lent)
add_executable(s1lent-server apps/server.cpp)
target_link_libraries(s1lent-server PRIVATE s1lent)
add_executable(s1lent-bench benchmarks/crypto_bench.cpp)
target_link_libraries(s1lent-bench PRIVATE s1lent)

if(S1LENT_BUILD_FUZZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "S1LENT_BUILD_FUZZERS requires Clang/libFuzzer")
  endif()
  foreach(target IN ITEMS packet route)
    add_executable(s1lent-fuzz-${target} fuzz/${target}_fuzz.cpp)
    target_link_libraries(s1lent-fuzz-${target} PRIVATE s1lent)
    target_compile_options(s1lent-fuzz-${target} PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_options(s1lent-fuzz-${target} PRIVATE -fsanitize=fuzzer,address,undefined)
  endforeach()
endif()

if(S1LENT_BUILD_TESTS)
  enable_testing()
  add_executable(s1lent-tests tests/unit.cpp)
  target_include_directories(s1lent-tests PRIVATE apps)
  target_link_libraries(s1lent-tests PRIVATE s1lent)
  add_test(NAME s1lent-unit COMMAND s1lent-tests)
  add_executable(s1lent-integration tests/integration.cpp)
  target_link_libraries(s1lent-integration PRIVATE s1lent)
  add_test(NAME s1lent-multinode COMMAND s1lent-integration)
  set_tests_properties(s1lent-multinode PROPERTIES TIMEOUT 15)
  add_executable(s1lent-handshake-tests tests/handshake.cpp)
  target_link_libraries(s1lent-handshake-tests PRIVATE s1lent)
  add_test(NAME s1lent-tls-handshake COMMAND s1lent-handshake-tests)
  set_tests_properties(s1lent-tls-handshake PROPERTIES TIMEOUT 15)
endif()
